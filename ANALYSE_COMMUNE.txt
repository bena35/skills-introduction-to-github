            "N° Secteur",
            "Mode légal soins",
            "Date de l'acte",
            "Nombre d'acte"
        },
        MissingField.UseNull
    ),

    // Secours Année RAA : à partir de Date de l'acte si Année absente/null
    RAA_Année_Secours = Table.AddColumn(
        RAA_Sélection,
        "Année_calc",
        each
            if [Année] <> null then [Année]
            else if [#"Date de l'acte"] <> null then Date.Year([#"Date de l'acte"])
            else null,
        Int64.Type
    ),
    RAA_Avec_Année = Table.RenameColumns(
        Table.RemoveColumns(RAA_Année_Secours, {"Année"}),
        {{"Année_calc", "Année"}}
    ),

    // Ajout des 4 colonnes absentes de RAA (valeurs nulles)
    RAA_Avec_Dates = Table.AddColumn(

            "Date début séquence", each null, type date),
        "Date fin séquence", each null, type date),

    RAA_Avec_Mesures = Table.AddColumn(
        Table.AddColumn(
            Table.AddColumn(
                RAA_Avec_Dates,
                "Nombre de jours", each null, Int64.Type),
            "Nombre de demi-journées", each null, Int64.Type),
        "Total en journées", each null, type number),

    RAA_Tagué = Table.AddColumn(RAA_Avec_Mesures, "Source", each "RAA", type text),


    // ════════════════════════════════════════════════════════════════════════
    // TABLE RPS
    // Contient : "Date entrée séjour", "Date sortie séjour",

    RPS_Sélection = Table.SelectColumns(
        RPS,
        {
            "Année",
            "Finess juridique",

            "N° Format",
            "N°IPP",
            "Nom",
            "Prenom",
            "Date naissance",
            "Âge à l'entrée",
            "Classe d'âge",
            "Sexe",

            "Code postal",
            "Département (libellé)",
            "Forme d'activité",
            "CodeForme",
            "Forme d'activité (libellé)",
            "N° UM",
            "N° Secteur",
            "Mode légal soins",
            "Date entrée séjour",
            "Date sortie séjour",
            "Date début séquence",
            "Date fin séquence",
            "Nombre de jours",
            "Nombre de demi-journées",
            "Total en journées"
        },
        MissingField.UseNull
    ),

    // Harmonisation des noms RPS → RAA
    RPS_Renommé = Table.RenameColumns(RPS_Sélection, {
        {"N°IPP", "N° IPP"},
        {"Âge à l'entrée", "Âge"}
    }),

    // Secours Année RPS : à partir de Date fin séquence si Année absente/null
    RPS_Année_Secours = Table.AddColumn(
        RPS_Renommé,
        "Année_calc",
        each
            if [Année] <> null then [Année]
            else if [#"Date fin séquence"] <> null then Date.Year([#"Date fin séquence"])
            else null,
        Int64.Type
    ),
    RPS_Avec_Année = Table.RenameColumns(
        Table.RemoveColumns(RPS_Année_Secours, {"Année"}),
        {{"Année_calc", "Année"}}

    ANALYSE_COMMUNE = Table.Combine({RAA_Tagué, RPS_Tagué}),

    RésultatTypé = Table.TransformColumnTypes(ANALYSE_COMMUNE, {

        {"N° Format", type text},
        {"N° IPP", type text},
        {"Nom", type text},
        {"Prenom", type text},
        {"Date naissance", type date},

        {"Classe d'âge", type text},

        {"Sexe_Libellé", type text},
        {"Code postal", type text},
        {"Département (libellé)", type text},

        {"CodeForme", type text},
        {"Forme d'activité (libellé)", type text},
        {"N° UM", type text},
        {"N° Secteur", type text},
        {"Mode légal soins", type text},
        {"Date de l'acte", type date},
        {"Nombre d'acte", Int64.Type},

        {"Date sortie séjour", type date},
        {"Date début séquence", type date},

    // ─── Ordre final (facultatif) ────────────────────────────────────────────
    Résultat = Table.ReorderColumns(RésultatTypé, {
        "Année",
        "Finess juridique",

        "Date début séquence",

        "Source"
    })
